cmake_minimum_required(VERSION 3.20)
project(IndependentMDX VERSION 0.33.5 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8 /W4 /EHsc)
  add_link_options($<$<CONFIG:Release>:/DEBUG:NONE> $<$<CONFIG:Release>:/INCREMENTAL:NO>)
endif()
include(third_party/lhasa/sources.cmake)
add_library(mdx_core STATIC src/formats.cpp src/player.cpp src/vgm.cpp src/archive.cpp)
target_link_libraries(mdx_core PUBLIC lhasa)
if(NOT WIN32)
  # Supply an independently built 0.13+ C ABI library for non-Windows tests.
  set(INDEPENDENT_OPM_LIBRARY "" CACHE FILEPATH "Independent OPM shared library")
  if(NOT EXISTS "${INDEPENDENT_OPM_LIBRARY}")
    message(FATAL_ERROR "Set INDEPENDENT_OPM_LIBRARY to a built 0.13+ shared library")
  endif()
  target_link_libraries(mdx_core PUBLIC "${INDEPENDENT_OPM_LIBRARY}")
endif()
target_include_directories(mdx_core PUBLIC src independent-opm/include)
add_executable(independent-mdx src/cli.cpp)
target_link_libraries(independent-mdx PRIVATE mdx_core)
if(WIN32)
  target_link_libraries(independent-mdx PRIVATE winmm)
  enable_language(RC)
  add_executable(IndependentMDX WIN32 src/gui.cpp resources/app.rc)
  target_link_libraries(IndependentMDX PRIVATE mdx_core winmm comdlg32 shell32 user32 gdi32 advapi32 comctl32 shlwapi uxtheme gdiplus ole32 uuid)
  if(MINGW)
    target_link_options(independent-mdx PRIVATE -municode)
    target_link_options(IndependentMDX PRIVATE -municode)
  endif()
  # x64 MSVC builds ship the prebuilt x86 helper. Rebuild separately with -A Win32.
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    add_executable(IndependentMDX-X68Bridge32 src/x68_helper.cpp)
    if(MINGW)
      target_link_options(IndependentMDX-X68Bridge32 PRIVATE -municode)
    endif()
  endif()
  foreach(target independent-mdx IndependentMDX)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/bin/IndependentMDX-X68Bridge32.exe" "$<TARGET_FILE_DIR:${target}>/IndependentMDX-X68Bridge32.exe")
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/bin/independent_opm.dll" "$<TARGET_FILE_DIR:${target}>/independent_opm.dll")
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_SOURCE_DIR}/THIRD_PARTY_NOTICES.txt" "$<TARGET_FILE_DIR:${target}>/THIRD_PARTY_NOTICES.txt")
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(mrl-tests tests/mrl_tests.cpp)
  target_link_libraries(mrl-tests PRIVATE mdx_core)
  add_test(NAME mrl-tests COMMAND mrl-tests)
  if(WIN32)
    add_custom_command(TARGET mrl-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/bin/independent_opm.dll" "$<TARGET_FILE_DIR:mrl-tests>/independent_opm.dll")
  endif()
  add_executable(archive-tests tests/archive_tests.cpp)
  target_link_libraries(archive-tests PRIVATE mdx_core)
  add_test(NAME archive-tests COMMAND archive-tests)
  if(WIN32)
    add_custom_command(TARGET archive-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/bin/independent_opm.dll" "$<TARGET_FILE_DIR:archive-tests>/independent_opm.dll")
  endif()
  add_executable(vgz-export-tests tests/vgz_export_tests.cpp)
  target_link_libraries(vgz-export-tests PRIVATE mdx_core)
  add_test(NAME vgz-export-tests COMMAND vgz-export-tests)
  if(WIN32)
    add_custom_command(TARGET vgz-export-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/bin/independent_opm.dll" "$<TARGET_FILE_DIR:vgz-export-tests>/independent_opm.dll")
  endif()
  add_executable(vgm-tests tests/vgm_tests.cpp)
  target_link_libraries(vgm-tests PRIVATE mdx_core)
  add_test(NAME vgm-tests COMMAND vgm-tests)
  add_executable(core-tests tests/core_tests.cpp)
  target_link_libraries(core-tests PRIVATE mdx_core)
  # The regression suite keeps many synthesis fixtures on the stack. MSVC's
  # default 1 MiB reserve can fail with STATUS_STACK_OVERFLOW before reporting.
  # Apply the larger reserve only to this test executable.
  if(MSVC)
    target_link_options(core-tests PRIVATE /STACK:8388608)
  elseif(MINGW)
    target_link_options(core-tests PRIVATE -Wl,--stack,8388608)
  endif()
  if(WIN32)
    add_custom_command(TARGET vgm-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/bin/independent_opm.dll" "$<TARGET_FILE_DIR:vgm-tests>/independent_opm.dll")
    add_custom_command(TARGET core-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/bin/independent_opm.dll" "$<TARGET_FILE_DIR:core-tests>/independent_opm.dll")
  endif()
  add_test(NAME core-tests COMMAND core-tests)
  if(WIN32)
    add_executable(gui-tests tests/gui_tests.cpp)
    target_link_libraries(gui-tests PRIVATE mdx_core winmm comdlg32 shell32 user32 gdi32 advapi32 comctl32 shlwapi uxtheme gdiplus ole32 uuid)
    if(MINGW)
      target_link_options(gui-tests PRIVATE -municode)
    endif()
    add_test(NAME gui-tests COMMAND gui-tests)
  endif()
endif()
